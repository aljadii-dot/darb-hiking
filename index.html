  <script>

    let watchId = null, recording = false, paused = false, points = [], totalDist = 0;

    const dial = document.getElementById('dial'), headingText = document.getElementById('headingText'), dirText = document.getElementById('dirText');

    const dirs = ['شمال', 'شمال شرق', 'شرق', 'جنوب شرق', 'جنوب', 'جنوب غرب', 'غرب', 'شمال غرب'];


    function updateCompass(deg) {

      const norm = ((deg % 360) + 360) % 360;

      dial.style.transform = `rotate(${-norm}deg)`;

      headingText.textContent = `${Math.round(norm)}°`;

      dirText.textContent = dirs[Math.round(norm / 45) % 8];

    }


    document.getElementById('simHeading').addEventListener('input', e => updateCompass(+e.target.value));


    document.getElementById('enableCompassBtn').addEventListener('click', async () => {

      if (typeof DeviceOrientationEvent !== 'undefined' && typeof DeviceOrientationEvent.requestPermission === 'function') {

        const res = await DeviceOrientationEvent.requestPermission();

        if (res !== 'granted') return alert('تم رفض إذن البوصلة');

      }

      window.addEventListener('deviceorientationabsolute', e => { if (e.alpha != null) updateCompass(360 - e.alpha); }, true);

      window.addEventListener('deviceorientation', e => {

        if (e.webkitCompassHeading != null) updateCompass(e.webkitCompassHeading);

        else if (e.alpha != null) updateCompass(360 - e.alpha);

      }, true);

    });


    function haversine(a, b) {

      const R = 6371000, toRad = x => x * Math.PI / 180;

      const dLat = toRad(b.lat - a.lat), dLon = toRad(b.lon - a.lon);

      const s = Math.sin(dLat/2)**2 + Math.cos(toRad(a.lat))*Math.cos(toRad(b.lat))*Math.sin(dLon/2)**2;

      return 2 * R * Math.atan2(Math.sqrt(s), Math.sqrt(1-s));

    }


    const canvas = document.getElementById('trackCanvas'), ctx = canvas.getContext('2d');

    function drawTrack() {

      ctx.clearRect(0, 0, canvas.width, canvas.height);

      if (points.length < 2) return;

      const lats = points.map(p => p.lat), lons = points.map(p => p.lon);

      const minLat = Math.min(...lats), maxLat = Math.max(...lats), minLon = Math.min(...lons), maxLon = Math.max(...lons);

      const pad = 20, spanLat = (maxLat - minLat) || 0.0001, spanLon = (maxLon - minLon) || 0.0001;

      ctx.strokeStyle = '#38bdf8'; ctx.lineWidth = 3; ctx.beginPath();

      points.forEach((p, i) => {

        const x = pad + ((p.lon - minLon) / spanLon) * (canvas.width - pad * 2);

        const y = canvas.height - pad - ((p.lat - minLat) / spanLat) * (canvas.height - pad * 2);

        if (i === 0) ctx.moveTo(x, y); else ctx.lineTo(x, y);

      });

      ctx.stroke();

    }


    function onPos(pos) {

      const pt = { lat: pos.coords.latitude, lon: pos.coords.longitude, alt: pos.coords.altitude, t: Date.now() };

      document.getElementById('lat').textContent = pt.lat.toFixed(6);

      document.getElementById('lon').textContent = pt.lon.toFixed(6);

      document.getElementById('alt').textContent = pt.alt != null ? pt.alt.toFixed(1) : 'غير متوفر';

      if (recording && !paused) {

        if (points.length) totalDist += haversine(points[points.length - 1], pt);

        points.push(pt);

        document.getElementById('dist').textContent = Math.round(totalDist);

        drawTrack();

      }

    }


    if (navigator.geolocation) {

      watchId = navigator.geolocation.watchPosition(onPos, () => {}, { enableHighAccuracy: true, maximumAge: 1000 });

    }


    const startBtn = document.getElementById('startBtn'), pauseBtn = document.getElementById('pauseBtn'), stopBtn = document.getElementById('stopBtn');

    startBtn.onclick = () => {

      recording = true; paused = false; points = []; totalDist = 0;

      document.getElementById('dist').textContent = '0'; ctx.clearRect(0, 0, canvas.width, canvas.height);

      startBtn.disabled = true; pauseBtn.disabled = false; stopBtn.disabled = false;

    };

    pauseBtn.onclick = () => {

      paused = !paused;

      pauseBtn.textContent = paused ? 'استئناف' : 'إيقاف مؤقت';

    };

    stopBtn.onclick = () => {

      recording = false; paused = false; pauseBtn.textContent = 'إيقاف مؤقت';

      startBtn.disabled = false; pauseBtn.disabled = true; stopBtn.disabled = true;

      if (!points.length) return;

      const saved = JSON.parse(localStorage.getItem('darb_tracks') || '[]');

      saved.unshift({ name: `مسار ${new Date().toLocaleString('ar-SA')}`, dist: Math.round(totalDist), count: points.length, points });

      localStorage.setItem('darb_tracks', JSON.stringify(saved));

      renderSaved();

    };


    function exportGPX(track) {

      const trkpts = track.points.map(p => `<trkpt lat="${p.lat}" lon="${p.lon}">${p.alt != null ? `<ele>${p.alt}</ele>` : ''}<time>${new Date(p.t).toISOString()}</time></trkpt>`).join('');

      const gpx = `<?xml version="1.0" encoding="UTF-8"?><gpx version="1.1" creator="Darb"><trk><name>${track.name}</name><trkseg>${trkpts}</trkseg></trk></gpx>`;

      const a = document.createElement('a');

      a.href = URL.createObjectURL(new Blob([gpx], { type: 'application/gpx+xml' }));

      a.download = `${track.name}.gpx`;

      a.click();

    }


    function renderSaved() {

      const list = document.getElementById('savedList');

      const saved = JSON.parse(localStorage.getItem('darb_tracks') || '[]');

      list.innerHTML = saved.length ? '' : '<li class="sub">لا توجد مسارات محفوظة بعد</li>';

      saved.forEach((t, idx) => {

        const li = document.createElement('li');

        li.innerHTML = `<div><strong>${t.name}</strong><small>المسافة: ${t.dist} م • النقاط: ${t.count}</small></div>`;

        const btn = document.createElement('button');

        btn.className = 'secondary'; btn.style.flex = 'none'; btn.textContent = 'GPX';

        btn.onclick = () => exportGPX(saved[idx]);

        li.appendChild(btn); list.appendChild(li);

      });

    }

    renderSaved();
